Booking Alerts

Email Notifications

When a student submits a booking, an alert will be sent to your email address. This uses EmailJS — a free service that sends emails directly from the browser, no server needed.

Enable Email AlertsSend you an email every time a new booking is made
Sign up free at emailjs.com → Email Services → Add New Service → copy Service ID
In EmailJS → Email Templates → create a template, paste this in the body:

New booking from: {{student_name}}
Email: {{student_email}}
Phone: {{student_phone}}
Plan: {{plan}}
Schedule: {{schedule}}
Payment: {{payment_method}}
In EmailJS → Account → General → Public Key

Firebase Setup

Firestore Rules

Go to console.firebase.google.com → backstage-music → Firestore → Rules, delete everything, and paste:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Bookings: anyone can create, student can read own, admin can read/write/delete
    match /bookings/{b} {
      allow create: if true;
      allow read: if request.auth != null && (
        request.auth.token.email == resource.data.email
        || request.auth.token.email == 'willemgouws80@gmail.com'
      );
      allow update, delete: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Public: anyone can read courses and pricing/config
    match /courses/{c} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /siteConfig/{d} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Students: user can read/write their own profile, admin can read/write all
    match /students/{email} {
      allow read: if request.auth != null && (
        request.auth.token.email == email
        || request.auth.token.email == 'willemgouws80@gmail.com'
      );
      allow create, update: if request.auth != null;
      allow delete: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Invoices: student can read their own, admin full access
    match /invoices/{i} {
      allow read: if request.auth != null && (
        resource.data.email == request.auth.token.email
        || request.auth.token.email == 'willemgouws80@gmail.com'
      );
      allow write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Attendance: student can read their own, admin full access
    match /attendance/{a} {
      allow read: if request.auth != null && (
        resource.data.studentEmail == request.auth.token.email
        || request.auth.token.email == 'willemgouws80@gmail.com'
      );
      allow write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Student notes: student can read their own, admin full access
    match /studentNotes/{n} {
      allow read: if request.auth != null && (
        resource.data.studentEmail == request.auth.token.email
        || request.auth.token.email == 'willemgouws80@gmail.com'
      );
      allow write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Resources: anyone can read (authenticated), admin can write
    match /resources/{r} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }

    // Quotes: admin only
    match /quotes/{q} {
      allow read, write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }
  }
}
Storage Rules

Go to console.firebase.google.com → backstage-music → Storage → Rules, replace with:

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /resources/{allPaths=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
        && request.auth.token.email == 'willemgouws80@gmail.com';
    }
  }
}
