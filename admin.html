Booking Alerts

Email Notifications

When a student submits a booking, an alert will be sent to your email address. This uses EmailJS — a free service that sends emails directly from the browser, no server needed.

Enable Email AlertsSend you an email every time a new booking is made
Sign up free at emailjs.com → Email Services → Add New Service → copy Service ID
In EmailJS → Email Templates → create a template, paste this in the body:

New booking from: {{student_name}}
Email: {{student_email}}
Phone: {{student_phone}}
Plan: {{plan}}
Schedule: {{schedule}}
Payment: {{payment_method}}
In EmailJS → Account → General → Public Key

Firebase Setup

Firestore Rules

Go to console.firebase.google.com → backstage-music → Firestore → Rules, delete everything, and paste:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Public: anyone can create a booking, nobody can read/edit/delete
    match /bookings/{b} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // Public: anyone can read courses and pricing/config
    // Only authenticated admin can write
    match /courses/{c} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /siteConfig/{d} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Private: only authenticated admin
    match /quotes/{q} {
      allow read, write: if request.auth != null;
    }
    match /invoices/{i} {
      allow read, write: if request.auth != null;
    }
    match /studentNotes/{n} {
      allow read, write: if request.auth != null;
    }
    match /attendance/{a} {
      allow read, write: if request.auth != null;
    }
  }
}
